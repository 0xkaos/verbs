הוא כָּתַב מִכְתָּב אֲרֹךְ.

---

הם כָּתְבוּ סִפְרִים מְעַנְיְנִים.

---

אני לָמַדְתִּי עִבְרִית בָּאוּנִיבֶרְסִיטָה.

---

הילדים לוֹמְדִים בֵּית סֵפֶר.

---

היא קוֹרֵאת סֵפֶר חָדָשׁ.

---

אנחנו קָרָאנוּ אֶת הַמֶּסֶר.

---

היא שָׁרָה שִׁיר יָפֶה.

---

אנחנו נָשִׁיר בַּמַּסִּבָּה מָחָר.

---

הוא שָׁמַר עַל הַבַּיִת בַּלַּיְלָה.

---

אני שׁוֹמֵר עַל הַסוֹד שֶׁלָּךְ.

---

נִשְׁמוֹר עַל הַמָּסֹרֶת הַזּוֹ.

---

הילדים רָצִים בַּפָּארְק כָּל בּוֹקֶר.

---

היא רָצָה מָהֵר לַאֲוִוירוֹן.

---

אנחנו עוֹבְדִים בְּיַחַד עַל הַפְּרוֹיֶקְט.

---

הוא עָבַד בְּחֶבְרָה גְּדוֹלָה.

---

הילדים מְשַׂחֲקִים בַּחֲצֵר.

---

הוא שִׂחֵק בַּסֶּרֶט הַחָדָשׁ.

---

היא צָחֲקָה מִן הַבְּדִיחָה.

---

אנחנו צוֹחֲקִים יַחַד.

---

הוא שָׁתָה כוֹס מַיִם.

---

אנחנו שׁוֹתִים קָפֶה בַּבֹּקֶר.

---

היא רָאֲתָה סֶרֶט מְעֻלֶּה.

---

האם אתה רוֹאֶה אֶת הַכּוֹכָבִים?

---

הוא פָּעַל בְּמַהֲלַךְ הַמִּשְׁבָּר.

---

אנחנו פּוֹעֲלִים לְמַעַן הַסְּבִיבָה.

---

אנחנו אוֹכְלִים אֲרוּחָה בְּיָחַד.

---

הוא אָכַל בַּמִּסְעָדָה הַחֲדָשָׁה.

---

אני הוֹלֵךְ לַחֲנוּת.

---

היא הָלְכָה לְבַקֵּר אֶת סַבְתָּהּ.

---

אני יוֹדֵעַ אֶת הַתְּשׁוּבָה.

---

היא יָדְעָה אֶת הַסּוֹד.

---

אני מֵבִין אֶת הַמַּצָּב.

---

היא לֹא הֵבִינָה אֶת הַשְּׁאֵלָה.

---

היא אָמְרָה שֶׁהִיא תָּבוֹא.

---

מה אַתָּה אוֹמֵר?

---

האם תָּבוֹא לַמְּסִבָּה?

---

הם בָּאוּ מִמְּקוֹמוֹת רְחוֹקִים.

---

אני חוֹשֵׁב שֶׁזֶּה רַעְיוֹן טוֹב.

---

היא חָשְׁבָה עַל הַהַצָּעָה.

---

היא אוֹהֶבֶת שׁוֹקוֹלָד.

---

הם אָהֲבוּ לְטַיֵּל בַּטֶּבַע.

---

אני רוֹצֶה לְלֶכֶת לַסֶּרֶט.

---

מה אַתָּה רוֹצֶה לֶאֱכֹל?

---

היא נָתְנָה לוֹ מַתָּנָה יָפָה.

---

אפשר לִתֵּן עֲזוֹר לַצְּווּת.

---

אני פּוֹגֵשׁ אֶת חֲבֵרַי בְּקָפֶה.

---

היא פָּגְשָׁה אוֹתוֹ בַּרְחוֹב.

---

הילד יָשַׁן כָּל הַלַּיְלָה.

---

אנו יְשֵׁנִים בְּאוֹהֶל בַּטֶּבַע.

---

אֲנִי מִשְׁתַּמֵּשׁ בַּמַּחְשֵׁב לַעֲבוֹדָה.

---

הם הִשְׁתַּמְּשׁוּ בַּמַּפָּה לְהִתְמַצֵּא בָּעִיר.

---

הם נִכְנְסוּ לַבַּיִת לְהִתְחַמֵּם.

---

בְּבַקָּשָׁה לֹא לְהִכָּנֵס לְחֶדֶר זֶה.

---

הוּא הִסְבִּיר אֶת הַחֹק לִתְּלַמִּידִים.

---

אֵינִי מֵבִין, אַתְּ יְכוֹלָה לְהַסְבִּיר לִי?

---

אני מַרְגִּישׁ שֶׁקַּר.

---

היא הִרְגִּישָׁה עֲיֵיפָה אַחֲרֵי הָרִיצָה.

---

הסרט הזה הִשְׁפִּיעַ עלי בצורה עמוקה.

---

הם מַשְׁפִּיעִים רבות על דעת הקהל.

---

הילד מִתְנַהֵג בְּאופֶן מְצוּיָן בַּגַּן.

---

אֲנַחנוּ צְרִיכִים לְהִתְנַהֵג בְּכָבוֹד זֶה לַזֶה.

---

אֲנִי מַאֲמִין בְּכִשְׁרוֹנוֹתֶיךָ.

---

הם הֶאֱמִינוּ שֶׁהַכֹּל יִהְיֶה בְּסֵדֶר.

---

אֲנִי מַכִּיר אוֹתוֹ מִבֵּית הַסֵּפֶר.

---

הִיא הִכִּירָה אֶת הַמָּקוֹם.

---

אֲנַחנוּ מַתְחִילִים אֶת הַשִּׁעוּר עַכְשָׁו.

---

הִיא הִתְחִילָה לִלְמוֹד עִבְרִית.

---

אֲנִי רוֹצֶה לְהִתְקַדֵּם בַּעֲבוֹדָה.

---

הַמְּפַתֵּחַ מִתְקַדֵּם בִּפְתּוּחַ הַתָּכְנִית.

---

הוּא הִפְסִיק לְדַבֵּר בַּשִּׁיעוּר.

---

אֲנִי מַפְסִיקָה לַעֲבוֹד בְּשָׁעָה שֵׁשׁ.

---

הוּא הִזְמִין אוֹתִי לְמְסִיבָה.

---

אֲנִי מַזְמִינָה אוֹכֶל בָּאִינְטֶרְנֵט.

---

אֲנִי מַשְׁאִיר אֶת הַמַּפְתְּחוֹת בַּשֻּׁלְחָן.

---

הִיא הִשְׁאִירָה לִי הוֹדָעָה.

---

אֲנִי מַצִּיעַ לָךְ לְהִתְחִיל לָלֶכֶת.

---

הוּא הִצִּיעַ לָנוּ רַעֲיוֹן חָדָשׁ.

---

אֲנִי מְחַפֵּשׂ עֲבוֹדָה חֲדָשָׁה.

---

הִיא חִפְּשָׂה אֶת הַמַּפְתֵּחוֹת בְּכָל הַבַּיִת.

---

הוּא הִשְׁלִים אֶת הַמַּסָּה בַּזְּמַן.

---

אֲנַחְנוּ צָרִיכִים לְהַשְׁלִים אֶת הַפְּרוֹיֶקְט.

---

אֲנִי מַבְטִיחַ לָבוֹא מֻקְדָּם מָחָר.

---

הִיא הִבְטִיחָה לִי עֶזְרָה בַּמִּבְחָן.

---

אֲנִי מְנַסֶּה לִלְמֹד עִבְרִית.

---

הֵם נִסּוּ לִפְתֹּר אֶת הַבְּעָיָה.

---

הוּא גִּילָּה תַּחֲרוּשֶׁת חֲדָשָׁה.

---

אֲנִי מְגַלֶּה עֲנָקִים בְּמַסָּעַי.

---

אֲנִי בּוֹחֵר לִלְמוֹד מוּסִיקָה.

---

הֵם בָּחֲרוּ בִּפְרוֹיֶקְט חָדָשׁ.

---

אֲנִי עוֹזֵר לַחֲבֵרִים שֶׁלִּי.

---

הִיא עָזְרָה לָנוּ בַּמַּטָּרָה.

---

אֲנִי רוֹצֶה לִשְׁאוֹל שְׁאֵלָה.

---

הַתַּלְמִידִים שׁוֹאֲלִים שְׁאֵלוֹת בַּשִּׁיעוּר.

---

אֲנִי זוֹכֵר אֶת הַיּוֹם הַזֶּה.

---

הִיא לֹא זָכְרָה אֶת הַשֵּׁם שֶׁלּוֹ.

---

אֲנִי רוֹצֶה לִקְנוֹת מְתָנָה לְחָבֵר.

---

הֵם קָנוּ בַּיִת חָדָשׁ.

---

הוּא פּוֹתֵחַ אֶת הַדֶּלֶת.

---

אַתָּה יָכוֹל לִפְתּוֹחַ אֶת הַתְּרוּפָה?

---

הוא סיפר לי סיפור מצחיק.

---

אני רוצה לספר לך את האמת.

---

אֲנִי מְדַמֶּה אֶת הַחֲלוֹם שֶׁלִּי כָּל יוֹם.

---

הֵם דִּמּוּ אֶת הָעֹלָם בְּלִי מִלְחָמוֹת.

---

אֲנִי מֵכִין אֶת הָאֹכֶל לָעֶרֶב.

---

הֵם הֵכִינוּ תּוֹכְנִית לַיּוֹם הֻלֶּדֶת.

---

אֲנִי מַמְצִיא סִיפּוּר חָדָשׁ לַיְּלָדִים.

---

הֵם הִמְצִיאוּ מָכוֹן לְחִסָּכוֹן בְּזֹמַן.

---

אני נָסַעְתִּי לְיוָּן בַּקַּיִץ.

---

הם נוֹסְעִים לְעבוֹדָה בָּרַכֶּבֶת.

---

הוּא בָּנָה בַּיִת גָּדוֹל בָּעִיר.

---

אֲנַחְנוּ בּוֹנִים גֶּשֶׁר חָדָשׁ עַכְשָׁו.

---

שִׁתַּפְתִּי אֶת הַחֲבֵרִים בַּרַעְיוֹן שֶׁלִּי.

---

הִיא מְשַׁתֶּפֶת אֶת כּוּלָם בַּהַחְלָטוֹת.

---

שִׁנִּיתִי אֶת דַּעְתִּי עַל הַנּוֹשֵׂא.

---

הֵם מְשַׁנִּים אֶת הַתָּכְנִית כָּל שָׁנָה.

---

הִצַּלְתִּי אֶת הַיֶּלֶד מֵהַמַּיִם.

---

הֵם מַצִּילִים אֶת הַפְּרוֹיֶקְט מִכִּשָּׁלוֹן.

---

הִסְכַּמְתִּי לָתֵת לוֹ עֶזְרָה.

---

הֵם מַסְכִּימִים עַל הַתָּכְנִית הַזּוֹ.

---

עָנִיתִי לַשְּׁאֵלָה שֶׁלּוֹ בְּמַהֵר.

---

הִיא עוֹנָה לְכָל הַשְּׁאֵלוֹת בַּכִּתָּה.

---

יָצַרְתִּי צִיּוּר יָפֶה בְּעֵט.

---

הֵם יוֹצְרִים מוּסִיקָה מְקוֹרִית.

---

הִצְטָרַפְתִּי לַקְּבוּצָה שֶׁלָּהֶם.

---

הִיא מִצְטָרֶפֶת לַמְּסִיבָּה מָחָר.

---

עָקַבְתִּי אַחֲרֵי הַמּוֹרֶה בַּדֶּרֶךְ.

---

הוּא עוֹקֵב אַחֲרֵי הַחֲדָשׁוֹת בַּיּוֹם.

---

הוֹבַלْתִּי אֶת הַקְּבוּצָה לַנִּצָּחוֹן.

---

הִיא מוֹבִילָה אֶת הַפְּרוֹיֶקְט בְּבִטָּחוֹן.

---

הָגַנְתִּי עַל הַיֶּלֶד מֵהַסַּכָּנָה.

---

הֵם מְגִינִים עַל הָעִיר בְּכָל הַלֵּב.

---

הוֹדַעְתִּי לָהֶם עַל הַשִּׁנוּי בַּתָּכְנִית.

---

הַמּוֹרָה מוֹדִיעָה לַהוֹרִים עַל הַצְלָחוֹת הַתַּלְמִידִים.

---

אוֹדִיעַ לְכֻלָּם אֶת הַחֲדָשׁוֹת בְּעוֹד שָׁעָה.

---

הִשְׁתַּתַּפְתִּי בַּתַּחֲרוּת הַשְּׁנָתִית.

---

הֵם מִשְׁתַּתְּפִים בְּפְגִישָׁה חָשׁוּבָה.

---

חָגַגְתִּי אֶת חַג הַפֶּסַח עִם מִשְׁפַּחְתִּי.

---

הֵם חוֹגְגִים אֶת יוֹם הָעַצְמָאוּת בָּרְחוֹבוֹת.

---

אָסַפְתִּי אֶת הַסְּפָרִים מֵהַשּׁוּלְחָן.

---

הֵם אוֹסְפִים תְּרוּמוֹת לְעוֹנִים.

---

חָלַמְתִּי עַל טִיּוּל לָעוֹלָם.

---

הִיא חוֹלֶמֶת עַל עָתִיד טוֹב יוֹתֵר.

---

הֵאַרְתִּי אֶת הַחֶדֶר בְּמֵנוֹרָה.

---

הַשֶּׁמֶשׁ מְאִירָה עַל הַרְרֵי יְרוּשָׁלַיִם.

---

הוּא מָכַר אֶת הַמְּכוֹנִית שֶׁלּוֹ.

---

הם מוֹכְרִים פֵּרוֹת בַּשּׁוּק.

---

בְּבַקָּשָׁה תְּתָאֵר אֶת הַבַּעְיָה.

---

הִיא תֵּאֲרָה אֶת הַנּוֹף בִּפְרָטִיּוּת.

---

בְּבַקָּשָׁה הָבֵא אֶת הַסֵּפֶר.

---

הִיא הֵבִיאָה עוּגָה לַמְּסִבָּה.

---

הוּא מְבַקֵּר אֶת הַהוֹרִים בְּכָל שַׁבָּת.

---

אָנוּ בִּקַּרְנוּ בְּמִוזֵיאוֹן הָאֻמָּנוּת.

---

אֲנִי צוֹפֶה בַּטֵּלֵוִיזְיָה כָּל עֶרֶב.

---

הִיא צָפְתָה בַּשֶּׁמֶשׁ הַזּוֹרַחַת.

---

אֲנִי נוֹהֵג לָעֲבוֹדָה כָּל בֹּקֶר.

---

הִיא נָהֲגָה בְּמֶשֶׁךְ שָׁעָה לְלֹא הַפְסָקָה.

אֲנִי תּוֹפֵס אֶת הַכַּדּוּר בְּיָד אַחַת.

---

הִיא תָּפְסָה אֶת הָרַעְיוֹן מַהֵר מְאוֹד.

---

אֲנִי מוֹשֵׁךְ אֶת הַחֶבֶל בְּכֹחַ.

---

הִיא מָשְׁכָה אֶת הַוִּילוֹן לְצַד.

---

אֲנִי מָרִים אֶת הַתִּיק הַכָּבֵד.

---

הֵם הֵרִימוּ אֶת הַדֶּגֶל בַּטֶּקֶס.

---

הַסֵּפֶר נִקְרָא בְּכָל הָעוֹלָם.

---

הוּא נִקְרָא לְשִׁמְשׁוֹן.

---

הַסֵּפֶר נִכְתַּב בְּשָׁנָה שֶׁעָבְרָה.

---

הַמִּכְתָּבִים נִכְתָּבִים בְּכָל יוֹם.

---

הַדּוּ''ח יִכָּתֵב עַד סוֹף הַחֹדֶשׁ.

---

הַמּוֹרָה הִכְתִּיבָה מִכְתָּב לַתַּלְמִידִים.

---

הַמְּנַהֵל מַכְתִּיב הוֹרָאוֹת לַמַּזְכִּירָה.

---

אני אַכְתִּיב לָךְ אֶת הַתּוֹכְנִית מָחָר.

---

הִקְרֵאתִי אֶת הַחֲבֵרִים לְמִסִּבַּת הַיּוֹם הֻלֶּדֶת.

---

הַמּוֹרֶה מַקְרִיא אֶת הַתַּלְמִיד לִקְרֹא אֶת הַשִּׁיר.

---

נַקְרִיא אֶת הַמִּשְׁפָּחָה לַחֲגִיגָה בְּיוֹם שִׁשִּׁי.

---

אנחנו מְדַבְּרִים עַל הַתָּכְנִית.

---

הוא דִּבֵּר בְּפִרְטֵי הַפְּרוֹיֶקְט.

---

הָעֲבוֹדָה נַעֲשְׂתָה בְּזֹאת עַל יְדֵי הַתַּלְמִידִים.

---

הַבֵּיצוֹת נַעֲשׂוֹת בְּמַחֲבַת בְּכָל בֹּקֶר.

---

הַתּוֹכְנִית תֵּעָשֶׂה אִם נִמְצָא אֶת הַזְּמַן הַמַּתְאִים.

---

הָרַעְיוֹן נֶחְשַׁב חָשׁוּב עַל יְדֵי כֻּלָּם.

---

הַתַּלְמִידִים נֶחְשָׁבִים חֲכָמִים בְּבֵית הַסֵּפֶר.

---

הַפִּתְרוֹן יֵחָשֵׁב נָכוֹן אִם נִבְדֹּק אוֹתוֹ.

---

הִשַּׁרְתִּי אֶת הַיְלָדִים לָשִׁיר שִׁירֵי חֲגִים.

---

הַמּוֹרָה מַשִּׁירָה אֶת הַתַּלְמִידִים בְּכָל שִׁעוּר.

---

אַשִּׁיר אֶת הַקָּהָל לָשִׁיר בְּטֶקֶס הַחֲגִיגָה מָחָר.

---

הֲרַצְתִּי אֶת הַיְלָדִים לְרוץ לְפָנַי בַּמֵּרוֹץ.

---

הַמּוֹרֶה מֵרִיץ אֶת הַתַּרְגִּילִים לְהִסָּיֵם בְּמָהִירוּת.

---

אָרִיץ אֶת הַמִּסְלוּל לְהֵרָצוֹת אִם נִרְצֶה לְנַצֵּחַ.

---

הִתְאַהַבְתִּי בָּהּ מֵרֶגַע שֶׁפָּגַשְׁתִּי אוֹתָהּ.

---

הוּא מִתְאַהֵב בְּכָל סֵפֶר חָדָשׁ שֶׁיּוֹצֵא.

---

אֶתְאַהֵב בַּמּוּסִיקָה אִם אֶשְׁמַע אוֹתָהּ שׁוּב.

---

נִפְגַּשְׁתִּי אִתּוֹ בְּבֵית הַקָּפֶה.

---

הֵם נִפְגָּשִׁים כָּל יוֹם שִׁשִּׁי.

---

אִם תִּרְצִי, נִפָּגֵשׁ מָחָר.

---

הִפָּגְשׁוּ בְּשָׁעָה שֶׁבַע!

---

נִבְחַרְתִּי לַתַּפְקִיד.

---

הוּא נִבְחָר לִהְיוֹת הַמַּנְהִיג.

---

אִם תִּבָּחֵר, תִּהְיֶה לָךְ הַהִזְדַּמְּנוּת.

---

הִבָּחֵר לַעֲבוֹדָה!

---

הוּא הִצְחִיק אֶת כֻּלָּם בְּסִפּוּרָיו.

---

הַסֶּרֶט מַצְחִיק מְאֹד.

---

אִם תִּרְצֶה, אַצְחִיק אוֹתְךָ.

---

הַצְחֵק אֶת הַיְלָדִים!

---

הַבַּיִת נִרְאָה מֵרָחוֹק.

---

הַכּוֹכָבִים נִרְאִים בַּלַּיְלָה.

---

אֵרָאֶה בַּתְּמוּנָה.

---

הֵרָאוּ בַּמָּקוֹם הַנָּכוֹן!

---

הוא הִפְעִיל אֶת הַמַּכְשִׁיר.

---

הַמְּנַהֵל מַפְעִיל אֶת הַתָּכְנִית הַחֲדָשָׁה.

---

אַפְעִיל אֶת הַמַּעֲרֶכֶת בְּעוֹד שָׁעָה.

---

הַפְעֵל אֶת הַמִּתְקָן!

---

נֶאֱמַר שֶׁהוּא יָבוֹא מָחָר.

---

הַדָּבָר נֶאֱמָר בְּבָרִיוּת.

---

יֵאָמֵר שֶׁזֶּה לֹא יָדוּעַ.

---

הֵאָמֵר בְּשֵׁם הַמְּנַהֵל!

---

הוּא נֶאֱמַן לִדְבָרוֹ.

---

הִיא נֶאֱמֶנֶת בְּעֵינֵי כֻּלָּם.

---

אֲנַחְנוּ נֵאָמֵן לְמִשְׁפָּחָה.

---

הֵאָמֵן לַחֲבֵרֶיךָ!

---

הוּא הִרְאָה לִי אֶת הַתְּמוּנָה.

---

הִיא מַרְאָה אֶת הַדֶּרֶךְ.

---

אַרְאֶה לְךָ אֶת הַסֶּרֶט.

---

הַרְאִי לִי אֶת הַסֵּפֶר!

---

הִפְגַּשְׁתִּי אֶת הַחֲבֵרִים.

---

הוּא מַפְגִּישׁ אֶת הַצְּוָתִים.

---

הוּא שִׁמֵּשׁ אֶת הַמֶּלֶךְ.

---

אֲנִי מְשַׁמֵּשׁ אֶת הַמַּכְשִׁיר.

---

הִכְנַסְתִּי אֶת הַצוֹעֲנִים.

---

הוּא מַכְנִיס אֶת הַחֲפָצִים.

---

הִיא סִדְּרָה אֶת הַחֶדֶר.

אֲנִי מְסַדֵּר אֶת הַסְּפָרִים.

נְסַדֵּר אֶת הַמִּסְמָכִים מָחָר.

---

סַדֵּר אֶת הַשֻּׁלְחָן!

---

הַשֶּׁמֶשׁ קָדְמָה לַיָּרֵחַ.

הוּא קוֹדֵם לִי בְּשָׁנָה.

אֲנִי אֶקְדֹּם אֶת הַפְּגִישָׁה.

קְדֹם לַפְּגִישָׁה!

הַגֶּשֶׁם פָּסַק פִּתְאוֹם.

הַמּוּזִיקָה פּוֹסֶקֶת כָּל לַיְלָה בְּשָׁעָה עֶשֶׂר.
