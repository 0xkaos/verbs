הִפְעַלְתֶּן אֶת הַמַּכְשִׁירִים בַּחֲדָרֵי הַמִשְׁחָק.

---

הם הִפְעִילוּ את המערכת החדשה בבוקר.

---

הִפְעִילוּ את המכשיר ברגע שהאורחים הגיעו.

---

מַפְעִיל את המכונה בכל בוקר.

---

היא מפעילה את המכונה בכל בוקר.

---

הם מפעילים את המערכת כדי לשפר את הביצועים.

---

כשהן מפעילות את המכשיר, הוא מתחיל לעבוד מיד.

---

אַפְעִיל את המכשיר מחר בבוקר.

---

אתה תפעיל את המכונה מחר בבוקר.

---

תַּפְעִילִי את המכשיר כשתגיעי הביתה.

---

כשהמתכנת יגיע, הוא יַפְעִיל את המערכת החדשה.

---

בבקשה תפעילי את המזגן כי חם מדי.

---

נַפְעִיל את המערכת ביום שני.

---

בבקשה, תפעילו את המכשיר כדי שנוכל לראות אם הוא עובד.

---

בבוקר תַּפְעֵלְנָה את המכונות החדשות בבית החרושת.

---

הם יַפְעִילוּ את המכונה בשעה תשע בבוקר.

---

הן תַּפְעֵלְנָה את המכונה בבוקר.

---

הַפְעֵל את המכשיר עכשיו.

---

הַפְעִילִי את המכשיר כדי להתחיל את התהליך.

---

בִּבְקָשָׁה, הַפְעִילוּ אֶת הַמַּזְגָּן לִפְנֵי שֶׁאֲנִי מַגִּיעָה.

---

הַפְעֵלְנָה עכשיו, בבקשה.

---

כדי לשפר את הביצועים של המחשב, עליך לְהַפְעִיל את התוכנות האחרונות.

---

בַּתְּחִלַת הַפַּגִישָׁה נֶאֱמַרְתִּי כִּי אֲנִי הַמַּנְחֶה הַחָדָשׁ.

---

אתמול נֶאֱמַרְתָּ לי שהכל בסדר, אבל היום אני רואה שהמצב שונה.

---

בַּחֲצָאי הַלַּיְלָה נֶאֱמַרְתְּ שֶׁתֵּצְאִי לִמְשִׁמְשֵׁי הַיָּרֵחַ.

---

נֶאֱמַר בַּתּוֹרָה שֶׁיִּשְׂרָאֵל יִזְכּוּ לְאֶרֶץ זָבַת חָלָב וּדְבַשׁ.

---

הדברים הללו נֶאֱמְרָה בוועידה האחרונה.

---

בִּמְשךְ הַדִּיּוּן, נֶאֱמַרְנוּ דְבָרִים חֲשׁוּבִים בְּמַאֲדֵי.

---

לאחר שהנושא נידון בהרחבה, נֶאֱמַרְתֶּם כי ההחלטה תתקבל בשבוע הבא.

---

בפגישה האחרונה נֶאֱמַרְתֶּן בבירור שהפרויקט יסתיים בזמן.

---

במהלך הדיון, נֶאֱמְרוּ דברים רבים על ידי כל המשתתפים.

---

המילים נֶאֱמְרוּ בבהירות ובנחת.

---

כך נֶאֱמָר בתורה

---

האמת נֶאֱמֶרֶת תמיד, גם אם לא תמיד שומעים אותה.

---

דברים נֶאֱמָרִים על ידי האנשים בחדר.

---

הדברים הנֶאֱמָרוֹת כאן חשובים מאוד.

---

אֵאָמֵר לוֹ שֶׁאֲנִי בָּא לְבִקְרוֹ מָחָר.

---

מחר תֵּאָמֵר דברים חשובים בפגישה.

---

אם לא תֵּאָמְרִי את האמת, לא אוכל לעזור לך.

---

בְּעוֹד כְּמָה יָמִים יֵאָמֵר שֶׁהַמֶּלֶךְ הוֹדִיעַ אֶת הַחֳלָטָתוֹ.

---

היא תֵּאָמֵר שֶׁהִיא עָיֵפָה מְאוֹד.

---

נֵאָמֵר לוֹ בְּבוֹקֶר שֶׁיִּזָּכֵּר לִקַּח אֶת הַמַּפְתֵּחוֹת.

---

תֵּאָמְרוּ לוֹ שֶׁאֲנִי אֱחַכֶּה לוֹ בַּבִּית.

---

מחר תֵּאָמַרְנָה כל הבנות שהן אוהבות את החג.

---

בְּיָמִים הָהֵם, יֵאָמְרוּ כָּל הָעַמִּים כִּי שָׁלוֹם יִהְיֶה בָּאָרֶץ.

---

בְּיַדֵי הַמּוֹרִים תֵּאָמַרְנָה הַדְּבָרִים לַתַּלְמִידִים.

---

הֵאָמֵר לוֹ שֶׁאֲנִי אֶפְגּשׁ אִתּוֹ בַּבּוֹקֶר.

---

הֵאָמְרִי לי את האמת.

---

הֵאָמְרוּ לוֹ בְּבִטָּחוֹן שֶׁיּוּכַל לְהַצְלִיחַ בְּכָל מִשִּׂימָה.

---

הֵאָמַרְנָה עכשיו, בבקשה.

---

המילים האלה צריכות לְהֵיאָמֵר בקול רם.

---

נֶאֱמַנְתִּי לְחֲבֵרִי בְּזִמָּנִים קָשִׁים.

---

כשאמרת שתהיה שם בשבילי, נֶאֱמַנְתָּ למילה שלך.

---

בכל הזמן הזה, נֶאֱמַנְתְּ לי ואף פעם לא בגדת בי.

---

הוא נֶאֱמַן לי כל חייו.

---

הָאֵם נֶאֱמְנָה לִבְנָהּ בְּכָל מַצָּב.

---

אנחנו נֶאֱמַנּוּ לדרכנו ולא סטינו מהמטרה.

---

לאחר שנים של חברות, נֶאֱמַנְתֶּם לי בכל פעם מחדש.

---

אַתֶּן נֶאֱמַנְתֶּן בְּמִשִּׁמְשׁוֹתֵיכֶם בְּנֶאֱמָנוּת לְכָל הַלֵּילוֹת.

---

הם נֶאֱמְנוּ לדרכם ולא סטו ממנה.

---

בְּמִשְׁפָּחָה זוֹ, הַהוֹרִים נֶאֱמְנוּ לִילָדֵיהֶם בְּכָל פַּעֲמִים.

---

הוא נֶאֱמָן לחבריו ותמיד שומר על הבטחותיו.

---

היא נֶאֱמֶנֶת בכל מצב.

---

אנחנו נֶאֱמָנִים לחברינו בכל מצב.

---

הן נֶאֱמָנוֹת לחברותיהן ותמיד שומרות על קשר.

---

אֵאָמֵן לך בכל ליבי.

---

אם תֵּאָמֵן בי, אני אוכיח לך שאתה יכול להצליח.

---

תֵּאָמְנִי בִּי גַּם בִּזְמַנִים קָשִׁים.

---

כשאני מסתכל על הציפורים, אני יֵאָמֵן לחלוטין למראה היפה של הטבע.

---

היא תֵּאָמֵן בו בכל ליבה

---

אנחנו נֵאָמֵן לך תמיד.

---

תֵּאָמְנוּ בַּתְּהִלִיךְ וְהַכּל יֵצֵא לְטוֹבָה.

---

תֵּאָמֵנָּה לוֹ בְּכָל לִבָּהּ.

---

הם יֵאָמְנוּ בדרכם ולא יסטו ממנה.

---

אנו מקווים שהן תֵּאָמֵנָּה לנו בכל מצב.

---

הֵאָמֵן בְּעַצְמְךָ וּבִכּוֹחוֹתֶיךָ.

---

הֵאָמְנִי בי, וְאֶהְיֶה לְצִדְּךָ תָמִיד.

---

הֵאָמְנוּ לוֹ, כִּי הוּא אֱמֶת.

---

הֵאָמֵנָּה עכשיו, בבקשה.

---

חשוב לְהֵאָמֵן לחבריך בכל מצב.

---

הִרְאֵיתִי לוֹ אֶת הַתְּמוּנָה הַיָּפָה שֶׁל הַשֶׁמֶשׁ הַשּׁוֹקַעַת.

---

הִרְאֵיתָ לִי אֶת הַדֶּרֶךְ הַחֲדָשָׁה בַּיַעֲרוֹת.

---

הִרְאֵיתְ לִי אֵיךְ לַעֲשׂוֹת אֶת הַמַּתְכוֹן הַחָדָשׁ.

---

הִרְאָה לִי אֵיךְ לִפְתּוֹר אֶת הַבְּעִיָה.

---

הִרְאֲתָה לִי אֶת הַדֶּרֶךְ הַחֲדָשָׁה בַּשְּׁכוּנָה.

---

הִרְאֵינוּ לוֹ אֲנַחְנוּ אֶת הַדֶּרֶךְ לַבֵּית הַסֵפֶר.

---

הִרְאֵיתֶם לוֹ אֶת הַדּוֹאָר שֶׁקִבַּלְתִי?

---

הִרְאֵיתֶן לוֹ אֶת הַתְּמוּנָה הַמּוּפְלָאָה שֶׁל הַשֶׁמֶשׁ בַּמַּעֲרָב.

---

הם הִרְאוּ לי את התמונות מהחופשה שלהם.

---

הִרְאוּ לָהֶם אֶת הַדֶּרֶךְ לַמּוּזֵיאוֹן.

---

כשהוא מראה לי את התמונות, אני מרגישה כאילו אני חוזרת בזמן.

---

היא מַרְאָה לי את הדרך.

---

הם מַרְאִים לנו את הדרך הנכונה.

---

כשאנחנו מראות לילדים את העולם, הם לומדים ממנו.

---

בבוקר אַרְאֶה לך את הגן היפה שלנו.

---

בוא נקווה שתַּרְאֶה לנו את הדרך הביתה.

---

תַּרְאִי לוֹ אֵיךְ עוֹשִׂים אֶת הַמָּטָלָה.

---

ביום שני הוא יַרְאֶה לי את הדרך החדשה לעבודה.

---

בבקשה, תַּרְאֶה לי איך עושים את זה.

---

נַרְאֶה לך את הדרך הביתה.

---

כשתפגשו את המנהל, תַּרְאוּ לו את ההצעה שלנו.

---

תַּרְאֶינָה לַיְלָדִים אֵיךְ לְהַכִּין עוּגָה בַּחֲגִיגָה.

---

הם יַרְאוּ לנו את הדרך החדשה בשבוע הבא.

---

הֵן תַּרְאֶינָה לַיְלָדִים אֶת הַסְרֵט הַחָדָשׁ.

---

הַרְאֵה לי איפה אתה מחביא את המתנות.

---

הַרְאִי לוֹ אֵיךְ עוֹשִׂים אֶת הַמַּטְבָּעוֹת.

---

הַרְאוּ לִי אֶת הַתִּמּוּנָה שֶׁלָּכֶם

---

הַרְאֶינָה עכשיו, בבקשה.

---

אנחנו רוצים לְהַרְאוֹת לך את הגן החדש.

---

הִפְגַּשְׁתִּי בֵּין שְׁנֵי הַחֲבֵרִים שֶׁלִּי בְּבִטְחָה.

---

הִפְגַּשְׁתָּ אוֹתִי עִם חָבֵר שֶׁלְּךָ מֵהָעֲבוֹדָה.

---

הִפְגַּשְׁתְּ אוֹתוֹ עִם חֲבֵרָהּ מֵהָעֲבוֹדָה.

---

הִפְגִּישׁ אוֹתִי עִם הַמּוֹרֶה שֶׁלִּי.

---

הִיא הִפְגִּישָׁה אוֹתָנוּ עִם הַמּוֹרֶה בְּשָׁעָה שֶׁהָיְתָה נוֹחָה לְכֻלָּנוּ.

---

הוא מַפְגִּישׁ אותנו בכל יום שני.
